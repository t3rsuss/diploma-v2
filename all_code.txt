def window_map(data):
    n = len(data)
    step = WIN
    if n // step > 400:
        step = (n // 400 // 512 + 1) * 512
    starts = list(range(0, max(n - 511, 1), step))
    X = np.array([feats(data[o:o + WIN]) for o in starts])
    proba = _fragment().model.predict_proba(X)
    return starts, step, proba.argmax(1), proba.max(1), X[:, 256]


def colors_for(classes):
    cm = plt.get_cmap("hsv", len(classes))
    return [matplotlib.colors.to_hex(cm(i)) for i in range(len(classes))]


def draw_strip(idx, classes, width=100):
    n = len(idx)
    rows = int(np.ceil(n / width))
    arr = np.full(rows * width, -1)
    arr[:n] = idx
    arr = arr.reshape(rows, width) + 1
    cmap = ListedColormap(["#ffffff"] + colors_for(classes))
    fig, ax = plt.subplots(figsize=(10, max(0.8, rows * 0.18)))
    ax.imshow(arr, cmap=cmap, vmin=0, vmax=len(classes), aspect="auto", interpolation="nearest")
    ax.axis("off")
    fig.tight_layout()
    return fig

import importlib.util
import pathlib
import sys
from io import BytesIO

from kaitaistruct import KaitaiStream, KaitaiStruct

GEN = pathlib.Path(__file__).resolve().parents[1] / "kaitai_dbg"
PARSERS = {
    "png": ("png", "Png"), "gif": ("gif", "Gif"), "ico": ("ico", "Ico"), "wav": ("wav", "Wav"),
    "gz": ("gzip", "Gzip"), "zip": ("zip", "Zip"), "docx": ("zip", "Zip"), "xlsx": ("zip", "Zip"),
    "pptx": ("zip", "Zip"), "ttf": ("ttf", "Ttf"), "sqlite": ("sqlite3", "Sqlite3"),
}
_cache = {}


def available(fmt):
    return fmt in PARSERS and GEN.exists()


def _parser(fmt):
    if fmt not in _cache:
        mod, cls = PARSERS[fmt]
        if str(GEN) not in sys.path:
            sys.path.append(str(GEN))
        spec = importlib.util.spec_from_file_location("kd_" + mod, GEN / (mod + ".py"))
        m = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(m)
        _cache[fmt] = getattr(m, cls)
    return _cache[fmt]


def _union(intervals, n):
    iv = sorted((max(0, s), min(n, e)) for s, e in intervals)
    total, cs, ce = 0, None, None
    for s, e in iv:
        if e <= s:
            continue
        if ce is None or s > ce:
            if ce is not None:
                total += ce - cs
            cs, ce = s, e
        else:
            ce = max(ce, e)
    return total + (ce - cs if ce is not None else 0)


def describe(data, fmt, limit=40):
    Parser = _parser(fmt)
    obj = Parser(KaitaiStream(BytesIO(data)))
    obj._read()
    for pname, attr in type(obj).__dict__.items():
        if isinstance(attr, property) and not pname.startswith("_"):
            try:
                getattr(obj, pname)
            except Exception:
                pass
    fields = []
    for name, info in getattr(obj, "_debug", {}).items():
        val = getattr(obj, name, None)
        if "arr" in info and isinstance(val, list):
            for i, (item, ai) in enumerate(zip(val, info["arr"])):
                fields.append((f"{name}[{i}]", ai["start"], ai["end"], type(item).__name__))
        elif info.get("start") is not None and info.get("end") is not None:
            fields.append((name, info["start"], info["end"], type(val).__name__))
    fields.sort(key=lambda f: f[1])
    cov = _union([(s, e) for _, s, e, _ in fields], len(data)) / max(len(data), 1)
    return {"parser": f"{Parser.__module__}.{Parser.__name__}", "fields_total": len(fields),
            "coverage_top_level": round(cov, 4),
            "fields": [dict(name=n, start=s, end=e, type=t) for n, s, e, t in fields[:limit]]}

def analyze_full(path, force_regions=False):
    path = pathlib.Path(path)
    rep = analyze(path)
    fmt = rep["format"]
    data = path.read_bytes()[:MAX_BYTES]
    if fmt in ("pe", "elf") and not force_regions:
        rep["branch"] = "виконуваний файл"
        return rep
    if not force_regions and known.available(fmt):
        try:
            rep["structure"] = known.describe(data, fmt)
            rep["branch"] = "відомий формат: опис структури (Kaitai Struct)"
            return rep
        except Exception as e:
            rep["structure_error"] = f"{type(e).__name__}: {e}"
    rep["regions"] = segment(data, _fragment())
    if force_regions:
        rep["branch"] = "карта областей (примусово)"
    elif fmt in ("other", "unknown") or rep["confidence"] < 0.5:
        rep["branch"] = "невідомий формат: карта областей за вмістом"
    else:
        rep["branch"] = "відомий формат без готового опису: карта областей за вмістом"
    return rep

import numpy as np

from src.triage.fragment import WIN, feats

STEP, LAM, BONUS, SNAP_R = 2048, 24.0, 24.0, 6144
MAGIC = {
    "pdf": [b"%PDF-"], "ole2": [b"\xd0\xcf\x11\xe0\xa1\xb1\x1a\xe1"], "gif": [b"GIF8"],
    "jpg": [b"\xff\xd8\xff"], "gz": [b"\x1f\x8b\x08"], "rtf": [b"{\\rtf"],
    "html": [b"<!DOCTYPE", b"<!doctype", b"<html", b"<HTML"], "xml": [b"<?xml"],
}


def _occurrences(data, sig):
    i = data.find(sig)
    while i != -1:
        yield i
        i = data.find(sig, i + 1)


def _viterbi(lp, lam):
    T, K = lp.shape
    dp = lp[0].copy()
    back = np.zeros((T, K), dtype=int)
    for t in range(1, T):
        bp = int(dp.argmax())
        sw = dp[bp] - lam
        stay = dp >= sw
        back[t] = np.where(stay, np.arange(K), bp)
        dp = np.where(stay, dp, sw) + lp[t]
    path = [int(dp.argmax())]
    for t in range(T - 1, 0, -1):
        path.append(int(back[t][path[-1]]))
    return path[::-1]


def _snap(data, b, cls):
    sigs = MAGIC.get(cls)
    if not sigs:
        return b
    lo, hi = max(0, b - SNAP_R), min(len(data), b + SNAP_R)
    seg = data[lo:hi]
    cand = [lo + i for s in sigs for i in _occurrences(seg, s)]
    return min(cand, key=lambda p: abs(p - b)) if cand else b


def segment(data, fc, lam=LAM, bonus=BONUS, snap=True):
    n = len(data)
    if n == 0:
        return []
    nb = (n + STEP - 1) // STEP
    offs = [min(max(k * STEP - (WIN - STEP) // 2, 0), max(n - WIN, 0)) for k in range(nb)]
    p = fc.model.predict_proba(np.array([feats(data[o:o + WIN]) for o in offs]))
    lp = np.log(p + 1e-6)
    if bonus:
        for c, sigs in MAGIC.items():
            if c not in fc.classes:
                continue
            ci = fc.classes.index(c)
            for s in sigs:
                for pos in _occurrences(data, s):
                    if pos // STEP < nb:
                        lp[pos // STEP, ci] += bonus
    path = _viterbi(lp, lam)
    segs, start = [], 0
    for k in range(1, nb + 1):
        if k == nb or path[k] != path[k - 1]:
            segs.append([start * STEP, min(k * STEP, n), path[start]])
            start = k
    if snap:
        for i in range(1, len(segs)):
            b = _snap(data, segs[i][0], fc.classes[segs[i][2]])
            if segs[i - 1][0] < b < segs[i][1]:
                segs[i][0] = segs[i - 1][1] = b
    out = []
    for s, e, ci in segs:
        k0, k1 = s // STEP, min((e - 1) // STEP + 1, nb)
        out.append(dict(start=s, end=e, length=e - s, cls=fc.classes[ci],
                        conf=float(p[k0:k1, ci].mean())))
    return out

import struct

AGREE_MIN = 0.6
X86_FAMILY = {3, 62}


def elf_machine(path):
    with open(path, "rb") as f:
        h = f.read(20)
    if len(h) < 20 or h[:4] != b"\x7fELF":
        return None
    fmt = "<H" if h[5] == 1 else ">H"
    return struct.unpack(fmt, h[18:20])[0]


def apply_rules(lvl, kind, pred, truth, machine=None):
    if kind != "ELF" or pred is None:
        return lvl, [], False

    reasons = []
    if truth is None:
        reasons.append("ELF без читаної таблиці секцій")
    elif machine in X86_FAMILY:
        agree = float((pred == truth).mean())
        if agree < AGREE_MIN:
            reasons.append(
                f"розмітка за вмістом збігається з таблицею секцій лише на {agree * 100:.1f} %"
            )

    raised = bool(reasons) and lvl == "low"
    return ("mid" if raised else lvl), reasons, raised

PRE_NAMES = {"elf": "ELF", "win64": "Win64", "dotnet": "Dot_Net",
             "win32": "Win32", "pe_generic": "PE"}
_CACHE = {}
_EXT = []


def _extractor():
    if not _EXT:
        _EXT.append(thrember.PEFeatureExtractor())
    return _EXT[0]


def file_kind(path):
    with open(path, "rb") as f:
        head = f.read(4)
    if head[:2] == b"MZ":
        return "PE"
    if head == b"\x7fELF":
        return "ELF"
    return None


def route(path):
    kind = file_kind(path)
    if kind is None:
        return None, None
    if kind == "ELF":
        return kind, "elf"
    try:
        pe = pefile.PE(str(path), fast_load=True)
        machine = pe.FILE_HEADER.Machine
        dirs = pe.OPTIONAL_HEADER.DATA_DIRECTORY
        dotnet = len(dirs) > 14 and dirs[14].VirtualAddress != 0
        pe.close()
    except Exception:
        return kind, "pe_generic"
    if dotnet:
        return kind, "dotnet"
    if machine == 0x8664:
        return kind, "win64"
    if machine == 0x14C:
        return kind, "win32"
    return kind, "pe_generic"


def _pre(name):
    key = "pre_" + name
    if key not in _CACHE:
        _CACHE[key] = lgb.Booster(model_file=str(PRE / f"EMBER2024_{name}.model"))
    return _CACHE[key]


def _own(key):
    if key not in _CACHE:
        metrics = {}
        mp = ROOT / "results" / f"detector_{key}_metrics.json"
        if mp.exists():
            metrics = json.loads(mp.read_text(encoding="utf-8"))
        _CACHE[key] = dict(
            booster=lgb.Booster(model_file=str(MODELS / f"detector_{key}.txt")),
            iso=joblib.load(MODELS / f"calibrator_{key}.joblib"),
            thr=json.loads((MODELS / f"thresholds_{key}.json").read_text(encoding="utf-8")),
            metrics=metrics,
        )
    return _CACHE[key]


def score_file(path):
    kind, key = route(path)
    if kind is None:
        return None
    with open(path, "rb") as f:
        data = f.read()
    vec = np.asarray(_extractor().feature_vector(data), dtype=np.float32).reshape(1, -1)
    pre = float(_pre(PRE_NAMES[key]).predict(vec)[0])
    out = dict(kind=kind, key=key, score_pre=pre)
    if key in ("elf", "win64"):
        m = _own(key)
        s = float(m["booster"].predict(vec)[0])
        out["score"] = s
        out["prob"] = float(m["iso"].predict([s])[0])
        out["t_mid"] = m["thr"]["thr_fpr_1"]
        out["t_high"] = m["thr"]["thr_fpr_01"]
        out["test_fpr_mid"] = m["metrics"].get("поріг FPR 1 %", {}).get("FPR на тесті")
        out["test_fpr_high"] = m["metrics"].get("поріг FPR 0,1 %", {}).get("FPR на тесті")
        out["source"] = "власна модель"
    else:
        out["score"] = pre
        out["t_mid"] = 0.2
        out["t_high"] = 0.5
        out["source"] = f"готова модель EMBER2024 ({PRE_NAMES[key]})"
    return out


def level(res):
    s = res["score"]
    if s >= res["t_high"]:
        return "high"
    if s >= res["t_mid"]:
        return "mid"
    return "low"

from pathlib import Path

import numpy as np

from src.parsing.parser import parse_binary

BLOCK = 4096


def block_entropies(arr: np.ndarray) -> np.ndarray:
    values = []
    for i in range(0, len(arr), BLOCK):
        blk = arr[i:i + BLOCK]
        if len(blk) < 256:
            continue
        counts = np.bincount(blk, minlength=256)
        p = counts[counts > 0] / len(blk)
        values.append(-(p * np.log2(p)).sum())
    return np.array(values) if values else np.zeros(1)


def extract(path) -> dict:
    path = Path(path)
    arr = np.frombuffer(path.read_bytes(), dtype=np.uint8)
    info = parse_binary(path)
    hist = np.bincount(arr, minlength=256) / max(len(arr), 1)
    be = block_entropies(arr)
    secs = info["sections"]

    row = {
        "file": info["file"],
        "format": info["format"],
        "size": len(arr),
        "file_entropy": info["file_entropy"],
        "blk_ent_mean": float(be.mean()),
        "blk_ent_std": float(be.std()),
        "blk_ent_min": float(be.min()),
        "blk_ent_max": float(be.max()),
        "frac_high_ent": float((be > 7.0).mean()),
        "zero_frac": float(hist[0]),
        "has_sections": int(len(secs) > 0),
        "n_sections": len(secs),
        "max_sec_entropy": max((s["entropy"] for s in secs), default=0.0),
        "n_imports": len(info["imports"]),
    }
    for i, v in enumerate(hist):

def _raw_features(self, bytez, pe):
    if pe is None:

    raw_obj = {
        "num_certs": 0,
        "self_signed": 0,
        "empty_program_name": 0,
        "no_countersigner": 0,
        "parse_error": 0,
        "chain_max_depth": 0,
        "latest_signing_time": 0,
        "signing_time_diff": 0,
    }
    try:
        signed_pe = SignedPEFile(io.BytesIO(bytez))
        for signed_data in signed_pe.iter_signed_datas():
            raw_obj["num_certs"] += 1
            if signed_data.signer_info.program_name is None:
                raw_obj["empty_program_name"] = 1

            countersigner = signed_data.signer_info.countersigner
            if countersigner is not None:
                signing_time = countersigner.signing_time.timestamp()
                if signing_time >= raw_obj["latest_signing_time"]:
                    raw_obj["latest_signing_time"] = signing_time
                raw_obj["signing_time_diff"] = signing_time - pe.FILE_HEADER.TimeDateStamp
            else:
                raw_obj["no_countersigner"] = 1

            certs = list(signed_data.certificates)
            if len(certs) > raw_obj["chain_max_depth"]:
                raw_obj["chain_max_depth"] = len(certs)
            for cert in certs[:-1]:
                if cert.issuer == cert.subject:
                    raw_obj["self_signed"] = 1
    except Exception:
        raw_obj["parse_error"] = 1
    return raw_obj

def byte_entropy(data: bytes) -> float:
    if not data:
        return 0.0
    counts = np.bincount(np.frombuffer(data, dtype=np.uint8), minlength=256)
    p = counts[counts > 0] / len(data)
    return float(-(p * np.log2(p)).sum())


def parse_binary(path) -> dict:
    path = Path(path)
    data = path.read_bytes()
    info = {
        "file": path.name,
        "size": len(data),
        "file_entropy": round(byte_entropy(data), 3),
        "format": None,
        "entrypoint": None,
        "sections": [],
        "imports": [],
        "error": None,
    }
    try:
        binary = lief.parse(str(path))
    except Exception as exc:
        info["error"] = str(exc)
        return info
    if binary is None:
        info["error"] = "lief: не вдалося розібрати"
        return info

    if isinstance(binary, lief.PE.Binary):
        info["format"] = "PE"
    elif isinstance(binary, lief.ELF.Binary):
        info["format"] = "ELF"
    else:
        info["format"] = "other"

    info["entrypoint"] = int(binary.entrypoint)
    info["sections"] = [
        {"name": s.name, "size": int(s.size), "entropy": float(s.entropy)}
        for s in binary.sections
    ]
    try:
        info["imports"] = [f.name for f in binary.imported_functions]
    except Exception:
        info["imports"] = []
    return info

import warnings

BLOCK = 512
PER_CLASS = 60
CLASSES = ["header", "code", "rodata", "data", "resources", "other"]
HEADER, CODE, RODATA, DATA, RES, OTHER = range(6)

_BASE = [f"h{i}" for i in range(64)] + [
    "entropy", "zeros", "ff", "printable", "high", "mean", "std", "uniq"
]
FEATURE_NAMES = (_BASE + [f"prev_{x}" for x in _BASE]
                 + [f"next_{x}" for x in _BASE] + ["rel_pos"])


def _enum_int(v):
    return int(getattr(v, "value", v))


def _pe_class(s):
    name = s.name.lower()
    ch = int(s.characteristics)
    if name == ".rsrc":
        return RES
    if name in (".reloc", ".debug"):
        return OTHER
    if ch & 0x20000000 or ch & 0x20:
        return CODE
    if ch & 0x80000000:
        return DATA
    return RODATA


def _elf_class(s):
    fl = int(s.flags)
    if fl & 4:
        return CODE
    if not fl & 2:
        return OTHER
    if fl & 1:
        return DATA
    return RODATA


def byte_labels(path, size):
    binary = lief.parse(str(path))
    if binary is None:
        return None
    if isinstance(binary, lief.PE.Binary):
        fmt, cls_fn = "PE", _pe_class
    elif isinstance(binary, lief.ELF.Binary):
        fmt, cls_fn = "ELF", _elf_class
    else:
        return None

    lab = np.full(size, OTHER, dtype=np.uint8)
    spans = []
    for s in binary.sections:
        off, sz = int(s.offset), int(s.size)
        if fmt == "ELF" and _enum_int(s.type) == 8:
            continue
        if sz <= 0 or off <= 0 or off >= size:
            continue
        spans.append((off, min(off + sz, size), cls_fn(s)))
    if not spans:
        return None
    first = min(a for a, _, _ in spans)
    lab[:first] = HEADER
    for a, b, c in spans:
        lab[a:b] = c
    return fmt, lab


def block_features(blocks):
    n = len(blocks)
    hist = np.zeros((n, 256), dtype=np.float32)
    for i in range(n):
        hist[i] = np.bincount(blocks[i], minlength=256)
    p = hist / BLOCK
    logp = np.zeros_like(p)
    np.log2(p, out=logp, where=p > 0)
    ent = -(p * logp).sum(1)
    h64 = p.reshape(n, 64, 4).sum(2)
    extra = np.stack([
        ent,
        p[:, 0],
        p[:, 255],
        p[:, 32:127].sum(1),
        p[:, 128:].sum(1),
        blocks.mean(1) / 255.0,
        blocks.std(1) / 255.0,
        (hist > 0).sum(1) / 256.0,
    ], axis=1)
    return np.hstack([h64, extra]).astype(np.float32)


def file_blocks(path, rng=None):
    rng = rng or np.random.default_rng(42)
    data = np.fromfile(str(path), dtype=np.uint8)
    n = len(data) // BLOCK
    if n < 2:
        return None
    res = byte_labels(path, len(data))
    if res is None:
        return None
    fmt, lab = res
    blocks = data[: n * BLOCK].reshape(n, BLOCK)
    lab_b = lab[: n * BLOCK].reshape(n, BLOCK)
    uniform = (lab_b == lab_b[:, :1]).all(axis=1)
    cls = lab_b[:, 0]

    keep = []
    for c in range(len(CLASSES)):
        idx = np.where(uniform & (cls == c))[0]
        if len(idx) > PER_CLASS:
            idx = rng.choice(idx, PER_CLASS, replace=False)
        keep.extend(idx.tolist())
    if not keep:
        return None
    keep = np.array(sorted(keep))

    prev = np.clip(keep - 1, 0, n - 1)
    nxt = np.clip(keep + 1, 0, n - 1)
    need = np.unique(np.concatenate([keep, prev, nxt]))
    f = block_features(blocks[need])
    ic, ip, inx = (np.searchsorted(need, a) for a in (keep, prev, nxt))
    rel = (keep / max(n - 1, 1)).astype(np.float32)[:, None]
    X = np.hstack([f[ic], f[ip], f[inx], rel])
    return fmt, X, cls[keep].astype(np.int64)

def feats(b):
    a = np.frombuffer(b, dtype=np.uint8)
    n = len(a)
    hist = np.bincount(a, minlength=256).astype(np.float32)
    p = hist / n
    nz = p[p > 0]
    ent = float(-(nz * np.log2(nz)).sum())
    chi = float(((hist - n / 256) ** 2 / (n / 256)).sum() / n)
    zr = len(zlib.compress(b, 1)) / n
    ent2, dif, ac, run = 0.0, 0.0, 0.0, 1.0
    if n > 1:
        pair = a[:-1].astype(np.int32) * 256 + a[1:]
        c2 = np.bincount(pair, minlength=65536)
        q = c2[c2 > 0] / (n - 1)
        ent2 = float(-(q * np.log2(q)).sum()) / 16
        dif = float(np.abs(np.diff(a.astype(np.int16))).mean()) / 255
        if a.std() > 0:
            ac = float(np.corrcoef(a[:-1], a[1:])[0, 1])
        cut = np.flatnonzero(a[1:] != a[:-1])
        run = float(np.diff(np.concatenate(([0], cut + 1, [n]))).max()) / n
    ll = np.array([len(x) for x in b.split(b"\n")], dtype=np.float32)
    extra = [ent, p[0], p[255], p[32:127].sum(), p[128:].sum(), a.mean() / 255,
             a.std() / 255, (hist > 0).sum() / 256, zr, chi, ent2, dif, ac, run,
             len(ll) / 100.0, ll.mean() / 100.0, ll.std() / 100.0]
    return np.concatenate([p, np.array(extra, dtype=np.float32)])


def windows(data, k, r):
    n = len(data)
    if n <= WIN:
        return [data]
    return [data[o:o + WIN] for o in (r.randrange(0, n - WIN + 1) for _ in range(k))]


def check_content(cls, b):
    if cls in SIG:
        return SIG[cls](b)
    try:
        t = b[:2_000_000].decode("utf-8")
    except UnicodeDecodeError:
        return False
    s = t.lstrip()
    if len(s) < 100 or "\x00" in t:
        return False
    try:
        if cls == "json":
            return s[0] in "{[" and isinstance(json.loads(t), (dict, list))
        if cls == "xml":
            ET.fromstring(t)
            return s.startswith("<")
        if cls == "html":
            return "<html" in s[:2000].lower() or "<!doctype html" in s[:2000].lower()
    except Exception:
        return False
    return s[0] not in "<{["


class FragmentClassifier:
    def __init__(self, path=MODEL_PATH):
        d = joblib.load(path)
        self.model = d["model"]
        self.classes = d["classes"]

    def classify(self, data, k=8, seed=0):
        wins = windows(data, k, random.Random(seed))
        proba = self.model.predict_proba(np.array([feats(w) for w in wins])).mean(axis=0)
        i = int(proba.argmax())
        return self.classes[i], float(proba[i]), dict(zip(self.classes, proba.tolist()))

def decode_text(b):
    if b[:3] == b"\xef\xbb\xbf":
        return b[3:].decode("utf-8"), "utf-8-bom"
    if b[:2] in (b"\xff\xfe", b"\xfe\xff"):
        return b.decode("utf-16"), "utf-16-bom"
    if len(b) >= 4 and b[0] != 0 and b[1] == 0 and b[3] == 0:
        return b.decode("utf-16-le"), "utf-16le"
    return b.decode("utf-8"), "utf-8"


def v_png(b):
    return b[:8] == b"\x89PNG\r\n\x1a\n" and b"IEND" in b[-16:]


def v_jpg(b):
    return b[:3] == b"\xff\xd8\xff" and b"\xff\xd9" in b[-64:]


def v_gif(b):
    return b[:6] in (b"GIF87a", b"GIF89a") and b[-1:] == b";"


def v_bmp(b):
    if b[:2] != b"BM" or len(b) < 26:
        return False
    size = struct.unpack("<I", b[2:6])[0]
    dib = struct.unpack("<I", b[14:18])[0]
    return size == len(b) and dib in (12, 40, 52, 56, 64, 108, 124)


def v_ico(b):
    if b[:4] != b"\x00\x00\x01\x00" or len(b) < 6:
        return False
    n = struct.unpack("<H", b[4:6])[0]
    return 1 <= n <= 64 and len(b) >= 6 + 16 * n


def v_pdf(b):
    return b[:5] == b"%PDF-" and b"%%EOF" in b[-1024:]


def v_zip(b):
    if b[:4] != b"PK\x03\x04":
        return False
    try:
        zipfile.ZipFile(io.BytesIO(b)).namelist()
        return True
    except Exception:
        return False


def v_gz(b):
    if b[:3] != b"\x1f\x8b\x08":
        return False
    try:
        zlib.decompressobj(31).decompress(b, 1_000_000)
        return True
    except Exception:
        return False


def v_7z(b):
    return b[:6] == b"7z\xbc\xaf\x27\x1c"


def v_sqlite(b):
    if b[:16] != b"SQLite format 3\x00" or len(b) < 100:
        return False
    ps = struct.unpack(">H", b[16:18])[0]
    return ps == 1 or (512 <= ps <= 32768 and ps & (ps - 1) == 0)


def v_wav(b):
    return b[:4] == b"RIFF" and b[8:12] == b"WAVE"


def v_mp3(b):
    return b[:3] == b"ID3" or (len(b) > 2 and b[0] == 0xFF and (b[1] & 0xE0) == 0xE0)


def v_pcap(b):
    return b[:4] in (b"\xd4\xc3\xb2\xa1", b"\xa1\xb2\xc3\xd4",
                     b"\x4d\x3c\xb2\xa1", b"\xa1\xb2\x3c\x4d", b"\x0a\x0d\x0d\x0a")


def v_ttf(b):

def v_pe(b):
    if b[:2] != b"MZ" or len(b) < 64:
        return False
    e = struct.unpack("<I", b[0x3C:0x40])[0]

def v_elf(b):

SKIP_EXT = {"zip", "xlsx", "pptx", "sqlite", "db", "lnk", "evtx", "elf", "so", ""}


def check(cls, b):
    if cls == "ole2":
        return b[:8] == bytes.fromhex("d0cf11e0a1b11ae1")
    if cls == "rtf":
        return b[:5] == b"{\\rtf"
    if cls == "pe":
        if b[:2] != b"MZ" or len(b) < 64:
            return False
        e = int.from_bytes(b[0x3C:0x40], "little")
        return e + 4 <= len(b) and b[e:e + 4] == b"PE\x00\x00"
    if cls == "docx":
        try:
            names = zipfile.ZipFile(io.BytesIO(b)).namelist()
            return "word/document.xml" in names
        except Exception:
            return False
    return check_content(cls, b)


def scan(root, seen):
    rows, rejected = [], collections.Counter()
    for p in sorted(root.rglob("*")):
        if not p.is_file():
            continue
        ext = p.suffix.lower().lstrip(".")
        if ext in SKIP_EXT:
            continue
        try:
            size = p.stat().st_size
            if size < 512 or size > MAXB:
                continue
            b = p.read_bytes()
        except OSError:
            continue
        h = hashlib.sha256(b).hexdigest()
        if h in seen:
            continue
        cls = EXT.get(ext)
        if cls is None:
            cls = "other"
        else:
            try:
                ok = check(cls, b)
            except Exception:
                ok = False
            if not ok:
                rejected[cls] += 1
                continue
        seen.add(h)
        rows.append([cls, h, size, ext, str(p)])
    return rows, rejected


def save(rows, name):
    out = BASE / "results" / name
    with open(out, "w", newline="", encoding="utf-8") as fh:
        w = csv.writer(fh)
        w.writerow(["format", "sha256", "size", "ext", "src"])
        w.writerows(rows)

def visit(val, parent, base, s, e, out, seen):

def covered(ranges, n):
    iv = sorted((max(0, s), min(n, e)) for s, e in ranges)
    iv = [(s, e) for s, e in iv if e > s]
    total, cs, ce = 0, None, None
    for s, e in iv:
        if ce is None or s > ce:
            if ce is not None:
                total += ce - cs
            cs, ce = s, e
        else:
            ce = max(ce, e)
    if ce is not None:
        total += ce - cs
    return total

def block_labels(ranges, n):
    nb = (n + BS - 1) // BS
    pb, sb = np.zeros(n, bool), np.zeros(n, bool)
    for s, e, o in ranges:
        s, e = max(0, s), min(n, e)
        if e > s:
            (pb if o else sb)[s:e] = True
    sb &= ~pb
    cov = pb | sb
    lab = np.full(nb, -1)
    for i in range(nb):
        a, b = i * BS, min(n, (i + 1) * BS)
        if cov[a:b].sum() >= 0.5 * (b - a):
            lab[i] = 1 if pb[a:b].sum() >= sb[a:b].sum() else 0
    return lab, float(cov.mean())


def block_feats(data):
    n = len(data)
    nb = (n + BS - 1) // BS
    pad = np.zeros(nb * BS, np.uint8)
    pad[:n] = np.frombuffer(data, np.uint8)
    valid = np.full(nb, BS, float)
    valid[-1] = n - (nb - 1) * BS
    idx = np.repeat(np.arange(nb), BS) * 256 + pad.astype(np.int64)
    cnt = np.bincount(idx, minlength=nb * 256).reshape(nb, 256).astype(float)
    cnt[-1, 0] -= BS - valid[-1]
    p = cnt / valid[:, None]
    ent = -(p * np.log2(np.where(p > 0, p, 1))).sum(1)
    asc = p[:, 32:127].sum(1) + p[:, [9, 10, 13]].sum(1)
    chi = ((cnt - valid[:, None] / 256) ** 2 / (valid[:, None] / 256)).sum(1) / valid
    diff = np.abs(np.diff(pad.reshape(nb, BS).astype(np.int16), axis=1)).mean(1)
    base = np.column_stack([ent, p[:, 0], asc, p[:, 128:].sum(1), chi, diff, p.reshape(nb, 16, 16).sum(2)])
    prev = np.vstack([base[:1], base[:-1]])
    nxt = np.vstack([base[1:], base[-1:]])
    pos = np.arange(nb)
    posf = np.column_stack([np.minimum(pos, 8), np.minimum(nb - 1 - pos, 8), pos / max(nb - 1, 1), np.full(nb, np.log2(nb + 1))])
    return np.hstack([base, prev[:, :6], nxt[:, :6], posf])

def smooth(lab):
    return np.array([int(round(np.mean(lab[max(0, i - 1):i + 2]))) for i in range(len(lab))])


def boundaries(lab):
    out, prev = [], None
    for i, v in enumerate(lab):
        if v < 0:
            continue
        if prev is not None and v != prev:
            out.append(i)
        prev = v
    return out


def match(tb, pb):
    used, tp = set(), 0
    for t in tb:
        cand = [(abs(p - t), j) for j, p in enumerate(pb) if j not in used]
        if cand:
            d, j = min(cand)
            if d <= TOL:
                used.add(j)
                tp += 1
    return tp

    return X, y

def stack(names):

    y = np.concatenate([l[l >= 0] for c in names for _, f, l in data_all[c]])
    return X, y


def f1_struct(y, p):
    tp = ((y == 0) & (p == 0)).sum()
    fp = ((y == 1) & (p == 0)).sum()
    fn = ((y == 0) & (p == 1)).sum()
    pr = tp / (tp + fp) if tp + fp else 0.0
    rc = tp / (tp + fn) if tp + fn else 0.0
    return 2 * pr * rc / (pr + rc) if pr + rc else 0.0


def smooth(lab):
    return np.array([int(round(np.mean(lab[max(0, i - 1):i + 2]))) for i in range(len(lab))])


def boundaries(lab):
    out, prev = [], None
    for i, v in enumerate(lab):
        if v < 0:
            continue
        if prev is not None and v != prev:
            out.append(i)
        prev = v
    return out


def match(tb, pb, tol):
    used, tp = set(), 0
    for t in tb:
        cand = [(abs(p - t), j) for j, p in enumerate(pb) if j not in used]
        if cand:
            d, j = min(cand)
            if d <= tol:
                used.add(j)
                tp += 1
    return tp


def prf(tp, npred, nt):
    p = tp / npred if npred else 0.0
    r = tp / nt if nt else 0.0
    return p, r, (2 * p * r / (p + r) if p + r else 0.0)


res = {}
for held in fmts:
    Xtr, ytr = stack([c for c in fmts if c != held])
    bestk = max(range(0, 8), key=lambda k: f1_struct(ytr, np.where(Xtr[:, -4] < k, 0, 1)))
    thr = max(np.arange(3.0, 8.0, 0.1), key=lambda t: f1_struct(ytr, np.where(Xtr[:, 0] < t, 0, 1)))
    kw = dict(n_estimators=200, num_leaves=15, learning_rate=0.08, subsample=0.8, subsample_freq=1,
              colsample_bytree=0.7, class_weight="balanced", random_state=SEED, verbose=-1, n_jobs=-1)
    m_all = lgb.LGBMClassifier(**kw).fit(Xtr, ytr)
    m_np = lgb.LGBMClassifier(**kw).fit(Xtr[:, :-4], ytr)
    pred = {
        "усе-наповн.": lambda f: np.ones(len(f), int),
        "голова K": lambda f: np.where(f[:, -4] < bestk, 0, 1),
        "ентропія": lambda f: np.where(f[:, 0] < thr, 0, 1),
        "модель": lambda f: m_all.predict(f).astype(int),
        "модель-без-поз.": lambda f: m_np.predict(f[:, :-4]).astype(int),
    }
    blk = {m: ([], []) for m in METHODS}
    bnd = {m: {t: [0, 0, 0] for t in TOLS} for m in METHODS}
    share = [0, 0]
    for _, f, l in data_all[held]:
        v = l >= 0
        share[0] += (l[v] == 0).sum()
        share[1] += v.sum()
        tb = boundaries(l)
        for m in METHODS:
            p = pred[m](f)
            blk[m][0].append(l[v])
            blk[m][1].append(p[v])
            pb = boundaries(np.where(v, p if m in ("усе-наповн.", "голова K") else smooth(p), -1))
            for t in TOLS:
                bnd[m][t][0] += match(tb, pb, t)
                bnd[m][t][1] += len(pb)
                bnd[m][t][2] += len(tb)
    res[held] = dict(share=100 * share[0] / share[1], K=bestk, thr=thr,
                     sf1={m: f1_struct(np.concatenate(blk[m][0]), np.concatenate(blk[m][1])) for m in METHODS},
                     bf1={t: {m: prf(*bnd[m][t])[2] for m in METHODS} for t in TOLS})


def table(title, getter):
    print(f"\n{title}")
    print(f"{'формат':8s}{'структ.,%':>10s}" + "".join(f"{m:>17s}" for m in METHODS))
    acc = {m: [] for m in METHODS}
    for h in fmts:
        vals = getter(res[h])
        print(f"{h:8s}{res[h]['share']:10.1f}" + "".join(f"{vals[m]:17.2f}" for m in METHODS))
        for m in METHODS:
            acc[m].append(vals[m])
    print(f"{'макро':8s}{'':10s}" + "".join(f"{np.mean(acc[m]):17.2f}" for m in METHODS))


table("F1 класу «структура» (за блоками по 256 Б)", lambda r: r["sf1"])
for t in TOLS:
    table(f"F1 меж між структурою і наповненням, допуск ±{t} блок(и)", lambda r, t=t: r["bf1"][t])
print("\nПідібрані параметри:", {h: (res[h]["K"], round(res[h]["thr"], 1)) for h in fmts})

def wilson(k, n):
    if n == 0:
        return float("nan"), float("nan")
    z = 1.96
    p = k / n
    d = 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return c - h, c + h

def wilson(k, n):
    if n == 0:
        return float("nan"), float("nan")
    z = 1.96
    p = k / n
    d = 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return c - h, c + h

def identify(data):
    try:
        r = m.identify_bytes(data)
        if r.ok:
            label = str(r.output.label)
            score = round(float(r.score), 3)
            dl = str(getattr(getattr(r, "dl", None), "label", ""))
        else:
            label, score, dl = "error", 0.0, ""
    except Exception:
        label, score, dl = "error", 0.0, ""
    try:
        mime = magic.from_buffer(data, mime=True)
        desc = magic.from_buffer(data)[:80]
    except Exception:
        mime, desc = "error", ""
    return label, dl, score, mime, desc


def variants(fmt, data, ext, sha):
    rng = random.Random(int(sha[:8], 16))
    size = len(data)
    out = [("orig", data, ext)]
    z = bytearray(data)
    z[:16] = b"\x00" * 16
    out.append(("zero_header", bytes(z), ext))
    r = bytearray(data)
    r[:16] = bytes(rng.randrange(256) for _ in range(16))
    out.append(("random_header", bytes(r), ext))
    out.append(("trunc_tail", data[: max(size // 2, 1)], ext))
    start = max(size // 2 - 2048, 0)
    out.append(("fragment", data[start:start + 4096], ext))
    if fmt in ("pe", "elf"):
        out.append(("spoof_jpg", data, "jpg"))
        out.append(("spoof_txt", data, "txt"))
    return out

    items = list(csv.DictReader(fh))
items.sort(key=lambda r: (r["format"], r["sha256"]))

last_fmt = None
with open(OUT, "w", newline="", encoding="utf-8") as fo:

                fo.flush()
                print(f"готово {last_fmt}, {time.time() - t0:.0f} с, рядків {n_rows}", flush=True)
            last_fmt = fmt
        data = src.read_bytes()
        for cond, d, shown in variants(fmt, data, ext, sha):
            label, dl, score, mime, desc = identify(d)

def feats(b):
    a = np.frombuffer(b, dtype=np.uint8)
    n = len(a)
    hist = np.bincount(a, minlength=256).astype(np.float32)
    p = hist / n
    nz = p[p > 0]
    ent = float(-(nz * np.log2(nz)).sum())
    chi = float(((hist - n / 256) ** 2 / (n / 256)).sum() / n)
    zr = len(zlib.compress(b, 1)) / n
    extra = [ent, p[0], p[255], p[32:127].sum(), p[128:].sum(), a.mean() / 255,
             a.std() / 255, (hist > 0).sum() / 256, zr, chi]
    return np.concatenate([p, np.array(extra, dtype=np.float32)])

            if len(test) >= target:

            if len(test) + len(parts[k]) > len(items) - 1:
                continue
            test += parts[k]
        if test and len(test) < len(items):
            ids = {id(x) for x in test}
            return [x for x in items if id(x) not in ids], test, "папки"
    sh = items[:]
    rng.shuffle(sh)
    return sh[target:], sh[:target], "файли"

def feats(b):
    a = np.frombuffer(b, dtype=np.uint8)

    hist = np.bincount(a, minlength=256).astype(np.float32)
    p = hist / n
    nz = p[p > 0]
    ent = float(-(nz * np.log2(nz)).sum())
    chi = float(((hist - n / 256) ** 2 / (n / 256)).sum() / n)
    zr = len(zlib.compress(b, 1)) / n
    ent2, dif, ac, run = 0.0, 0.0, 0.0, 1.0
    if n > 1:
        pair = a[:-1].astype(np.int32) * 256 + a[1:]
        c2 = np.bincount(pair, minlength=65536)

        ent2 = float(-(q * np.log2(q)).sum()) / 16
        dif = float(np.abs(np.diff(a.astype(np.int16))).mean()) / 255
        if a.std() > 0:
            ac = float(np.corrcoef(a[:-1], a[1:])[0, 1])
        cut = np.flatnonzero(a[1:] != a[:-1])
        run = float(np.diff(np.concatenate(([0], cut + 1, [n]))).max()) / n
    ll = np.array([len(x) for x in b.split(b"\n")], dtype=np.float32)
    extra = [ent, p[0], p[255], p[32:127].sum(), p[128:].sum(), a.mean() / 255,

             len(ll) / 100.0, ll.mean() / 100.0, ll.std() / 100.0]

def build(rng, pool, n_img):
    classes = list(pool)
    out = []
    for _ in range(n_img):
        seq, prev = [], None
        for _ in range(rng.randint(3, 6)):
            c = rng.choice([x for x in classes if x != prev])
            d = pathlib.Path(rng.choice(pool[c])).read_bytes()
            seq.append((c, d[:rng.randint(16384, min(len(d), 131072))]))

            s += len(d)

        nb = (n + STEP - 1) // STEP
        truth = [next(c for e, c in ends if min(k * STEP + STEP // 2, n - 1) < e) for k in range(nb)]
        wins = [data[min(max(k * STEP - (WIN - STEP) // 2, 0), max(n - WIN, 0)):][:WIN] for k in range(nb)]
        out.append(dict(truth=truth, true_b=[e for e, _ in ends[:-1]], wins=wins))

def viterbi(lp, lam):
    T, K = lp.shape
    dp = lp[0].copy()
    back = np.zeros((T, K), dtype=int)
    for t in range(1, T):
        bp = int(dp.argmax())
        sw = dp[bp] - lam
        stay = dp >= sw
        back[t] = np.where(stay, np.arange(K), bp)
        dp = np.where(stay, dp, sw) + lp[t]
    path = [int(dp.argmax())]
    for t in range(T - 1, 0, -1):
        path.append(int(back[t][path[-1]]))
    return [fc.classes[i] for i in path[::-1]]


def match(true_b, pred_b):
    used, tp = set(), 0
    for t in true_b:
        cand = [(abs(p - t), j) for j, p in enumerate(pred_b) if j not in used]
        if cand:
            d, j = min(cand)
            if d <= TOL:
                used.add(j)
                tp += 1
    return tp


def score(images, labeler):
    s = dict(ok=0, okf=0, bins=0, tp=0, np=0, nt=0)
    for im in images:
        lab = labeler(im)
        tr = im["truth"]
        s["bins"] += len(tr)
        s["ok"] += sum(a == b for a, b in zip(lab, tr))
        s["okf"] += sum(fam(a) == fam(b) for a, b in zip(lab, tr))
        pb = [k * STEP for k in range(1, len(lab)) if lab[k] != lab[k - 1]]
        s["tp"] += match(im["true_b"], pb)
        s["np"] += len(pb)
        s["nt"] += len(im["true_b"])
    p = s["tp"] / s["np"] if s["np"] else 0.0
    r = s["tp"] / s["nt"]
    f1 = 2 * p * r / (p + r) if p + r else 0.0
    return 100 * s["ok"] / s["bins"], 100 * s["okf"] / s["bins"], 100 * p, 100 * r, f1, s["np"], s["nt"]

head = f"{'метод':22s}{'байти,%':>9s}{'родина,%':>10s}{'точн.меж':>10s}{'повнота':>9s}{'F1':>7s}{'меж/істинних':>14s}"

    sigs = MAGIC.get(cls)

import collections
import pathlib
import random

from remotezip import RemoteZip

BASE = pathlib.Path(r"C:\Users\vlad\Desktop\diploma-v2")
GD = BASE / "data" / "govdocs"
URL = "https://downloads.digitalcorpora.org/corpora/files/govdocs1/zipfiles/{:03d}.zip"
USED = {0, 1, 2, 500}
EXT = {"jpg": "jpg", "jpeg": "jpg", "gif": "gif", "gz": "gz", "xml": "xml",
       "txt": "txt", "log": "log", "csv": "csv", "rtf": "rtf"}
DEV_Q = {"gif": 60, "jpg": 60, "gz": 40, "xml": 50, "txt": 50, "log": 40, "csv": 40, "rtf": 40}
TEST_Q = {"gif": 40, "jpg": 40, "gz": 30, "xml": 40, "txt": 40, "log": 30, "csv": 30, "rtf": 30}
N_THREADS, PER_THREAD, MAX_MEMBER, RESERVE = 36, 8, 10_000_000, 1.3
rng = random.Random(42)

cand = rng.sample([n for n in range(1000) if n not in USED], N_THREADS)
split = N_THREADS * 2 // 3
SIDES = {"dev": (cand[:split], DEV_Q), "test": (cand[split:], TEST_Q)}


def run(side, threads, quota):
    got = collections.Counter()
    for n in threads:
        try:
            with RemoteZip(URL.format(n)) as z:
                by = collections.defaultdict(list)
                for i in z.infolist():
                    if i.is_dir() or "." not in i.filename:
                        continue
                    cls = EXT.get(i.filename.rsplit(".", 1)[-1].lower())
                    if cls and 512 <= i.file_size <= MAX_MEMBER:
                        by[cls].append(i.filename)
                pick = []
                for cls, q in quota.items():
                    names = by.get(cls, [])
                    rng.shuffle(names)
                    k = min(PER_THREAD, max(0, int(q * RESERVE) - got[cls]), len(names))
                    pick += [(cls, x) for x in names[:k]]
                    got[cls] += k
                if not pick:
                    continue
                out = GD / f"{side}_extra" / f"{n:03d}"
                out.mkdir(parents=True, exist_ok=True)
                for cls, name in pick:
                    (out / pathlib.Path(name).name).write_bytes(z.read(name))
            print(f"{side} {n:03d}: {len(pick)} файлів", flush=True)
        except Exception as e:
            print(f"{side} {n:03d}: пропущено ({type(e).__name__}: {e})", flush=True)
    print(f"\n{side}: набрано", dict(got), "\n")


for side, (threads, quota) in SIDES.items():
    print(f"{side}: каталоги {sorted(threads)}")
    run(side, threads, quota)

import collections
import csv
import os
import pathlib
import random
import sys

import magic
from magika import Magika

random.seed(42)
SRC = [pathlib.Path(p) for p in sys.argv[1].split(";")]
OUT = pathlib.Path(sys.argv[2])

EXT = {"png", "jpg", "gif", "bmp", "ico", "pdf", "zip", "gz", "7z", "sqlite", "db",
       "wav", "mp3", "json", "xml", "html", "txt", "pcap", "ttf"}
PE_EXT = {"exe", "dll", "sys", "ocx"}
SKIP_DIRS = {"winsxs", "installer", "assembly", "servicing", "temp",
             "$recycle.bin", "node_modules", ".git"}
PER = 30
CAND = PER * 3
MAXB = 2_000_000
MAX_PER_ROOT = 60_000

found = collections.defaultdict(list)


def head4(p):
    try:
        with open(p, "rb") as f:
            return f.read(4)
    except OSError:
        return b""


for root_dir in SRC:
    is_elf_root = "elf_clean" in str(root_dir).lower()
    n = 0
    for dp, dn, fn in os.walk(root_dir, onerror=lambda e: None):
        dn[:] = [d for d in dn if d.lower() not in SKIP_DIRS]
        for name in fn:
            n += 1
            p = pathlib.Path(dp) / name
            ext = p.suffix.lower().lstrip(".")
            if is_elf_root:
                key = "elf" if ext == "" else None
            elif ext in PE_EXT:
                key = "pe"
            elif ext in EXT:
                key = ext
            else:
                key = None
            if key is None or len(found[key]) >= CAND:
                continue
            try:
                size = p.stat().st_size
            except OSError:
                continue
            if size < 512 or size > 50_000_000:
                continue
            h = head4(p)
            if key == "pe" and h[:2] != b"MZ":
                continue
            if key == "elf" and h != b"\x7fELF":
                continue
            found[key].append(p)
        if n > MAX_PER_ROOT:
            break
    print(f"обхід {root_dir}: переглянуто {n}, знайдено класів {len(found)}", flush=True)

m = Magika()
rows = []
for key, paths in sorted(found.items()):
    random.shuffle(paths)
    for p in paths[:PER]:
        try:
            data = p.read_bytes()[:MAXB]
        except OSError:
            continue
        shown = p.suffix.lower().lstrip(".")
        variants = [("orig", data, shown)]
        broken = bytearray(data)
        broken[:16] = b"\x00" * 16
        variants.append(("zero_header", bytes(broken), shown))
        if key in ("pe", "elf"):
            variants.append(("spoof_jpg", data, "jpg"))
        for cond, d, ext_shown in variants:
            r = m.identify_bytes(d)
            label = r.output.label if r.ok else "error"
            score = round(float(r.score), 3) if r.ok else 0.0
            rows.append([p.name, key, cond, ext_shown, label, score,
                         magic.from_buffer(d, mime=True)])
    print("готово:", key, min(len(paths), PER), flush=True)

OUT.parent.mkdir(parents=True, exist_ok=True)
with open(OUT, "w", newline="", encoding="utf-8") as fh:
    w = csv.writer(fh)
    w.writerow(["file", "true", "condition", "ext_shown", "magika", "magika_score", "libmagic_mime"])
    w.writerows(rows)

print("\nНайчастіші мітки Magika за істинним класом (стан orig):")
cnt = collections.defaultdict(collections.Counter)
for r in rows:
    if r[2] == "orig":
        cnt[r[1]][r[4]] += 1
for k, c in sorted(cnt.items()):
    print(f"{k:8s}", sum(c.values()), c.most_common(3))

files, seen = [], set()
src_count = collections.Counter()


def add(cls, data, source):
    h = hashlib.sha256(data).hexdigest()
    if h in seen:
        return
    seen.add(h)
    files.append((cls, data))
    src_count[source] += 1

    for it in by[c]:
        ext = it["ext"]
        p = CORPUS / c / (it["sha256"][:12] + ("." + ext if ext else ""))
        add(c, p.read_bytes(), "Windows")

ext_n = collections.Counter()
for d in sorted(EXT.iterdir()):
    cls = DIRMAP.get(d.name, d.name)
    if cls not in win_classes:
        continue
    for p in sorted(d.iterdir()):
        b = p.read_bytes()
        if len(b) < 512 or ext_n[cls] >= EXT_PER or not check_content(cls, b):
            continue
        ext_n[cls] += 1
        add(cls, b, "Ubuntu")

with open(GD_DEV, encoding="utf-8", newline="") as fh:
    gd = list(csv.DictReader(fh))
gby = collections.defaultdict(list)
for r in gd:
    gby[r["format"]].append(r)

for cls, data in files:
